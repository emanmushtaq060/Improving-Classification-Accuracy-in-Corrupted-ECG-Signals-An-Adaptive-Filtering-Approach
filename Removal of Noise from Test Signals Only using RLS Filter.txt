%% ================= Load ECG Data =================
load("C:\Program Files\KMSpico\temp\45aa0ded-d408-427c-8262-6d259ebd8db7_ECGData.zip.db7\ECGData.mat")

Signals = ECGData.Data;
Labels  = ECGData.Labels;

%% ================= Train / Test Split =================
Xtrain = Signals([1:85,97:115,127:151],:);
Ytrain = Labels([1:85,97:115,127:151],:);

Xtest  = Signals([86:96,116:126,152:162],:);
Ytest  = Labels([86:96,116:126,152:162],:);

%% ================= Add Noise to TEST DATA ONLY =================
snr = +10;   % Noise level in dB
Xtest_noisy = awgn(Xtest, snr, 'measured');

%% ================= RLS ADAPTIVE FILTER DENOISING =================
M = 5;              
lambda = 0.99;      
delta  = 0.01;      

%% ---- Denoise TEST data ----
Xtest_denoised = zeros(size(Xtest_noisy));

for i = 1:size(Xtest_noisy,1)
    d = Xtest(i,:)';          
    x = Xtest_noisy(i,:)';    

    w = zeros(M,1);
    P = (1/delta)*eye(M);
    y = zeros(length(x),1);

    for n = M:length(x)
        x_vec = x(n:-1:n-M+1);
        y(n) = w' * x_vec;
        e = d(n) - y(n);
        k = (P*x_vec) / (lambda + x_vec'*P*x_vec);
        w = w + k*e;
        P = (P - k*x_vec'*P) / lambda;
    end

    Xtest_denoised(i,:) = y';
end

Xtest = Xtest_denoised;

%% ================= PCA (TRAIN DATA ONLY) =================
[coeff, scoreTrain, ~, ~, explained, mu_pca] = pca(Xtrain);
scoreTrain95 = scoreTrain(:,1:2);

%% ================= Classifier (Neural Network) =================
%mdl = fitctree(scoreTrain95,Ytrain); % mdl is a ClassificationTree model.
%mdl = fitcdiscr(scoreTrain95,Ytrain,'DiscrimType', 'linear');
%mdl = fitcknn(scoreTrain95,Ytrain);
%mdl = fitcnb(scoreTrain95,Ytrain);
mdl = fitcensemble(scoreTrain95,Ytrain);
%mdl = fitcnet(scoreTrain95, Ytrain, ...
         %     'LayerSizes', [20], ...
          %    'Standardize', true);

%% ================= Test Projection =================
scoreTest95 = (Xtest - mu_pca) * coeff(:,1:2);

%% ================= Prediction =================
YTest_predicted = predict(mdl, scoreTest95);

Accuracy = sum(categorical(Ytest) == categorical(YTest_predicted)) ...
           * 100 / numel(Ytest)
