%% Load ECG Data
load("C:\Program Files\KMSpico\temp\66eba54d-b7f6-41dd-bc1a-0ae9e4162f15_ECGData.zip.f15\ECGData.mat")
Signals = ECGData.Data;     
Labels  = ECGData.Labels;   

%% Add Gaussian Noise (-10 dB SNR)
SNR = +10;
Signals_noisy = awgn(Signals, SNR, 'measured');

%% ========== RLS Adaptive Noise Canceller ==========
M = 20;            % Filter order
lambda = 0.99;     % Forge)tting factor (0.98–1)
delta = 0.01;      % Regularization constant

Signals_RLS = zeros(size(Signals));

for i = 1:size(Signals,1)

    s = Signals(i,:);           % Clean ECG (simulation only)
    d = Signals_noisy(i,:);     % Noisy ECG (desired signal)

    noise_ref = d - s;          % Reference noise
    N = length(d);

    % Initialization
    w = zeros(M,1);             % Filter weights
    P = (1/delta) * eye(M);     % Inverse correlation matrix
    y = zeros(1,N);
    e = zeros(1,N);

    % Zero padding
    noise_pad = [zeros(1,M-1), noise_ref];

    for n = 1:N
        x = noise_pad(n+M-1:-1:n)';   % Reference input

        y(n) = w' * x;                % Estimated noise
        e(n) = d(n) - y(n);           % Denoised ECG

        % RLS gain vector
        k = (P*x) / (lambda + x'*P*x);

        % Weight update
        w = w + k * e(n);

        % Inverse correlation update
        P = (P - k*x'*P) / lambda;
    end

    Signals_RLS(i,:) = e;
end

%% Train-Test Split
Xtrain = Signals_RLS([1:85,97:115,127:151],:);
Ytrain = Labels([1:85,97:115,127:151],:);

Xtest  = Signals_RLS([86:96,116:126,152:162],:);
Ytest  = Labels([86:96,116:126,152:162],:);

%% PCA (2 Components)
[coeff, scoreTrain, ~, ~, explained, mu_pca] = pca(Xtrain);
scoreTrain2 = scoreTrain(:,1:2);

%% Train Classifier (Ensemble)
%mdl = fitctree(scoreTrain2,Ytrain); % mdl is a ClassificationTree model.
%mdl = fitcdiscr(scoreTrain2,Ytrain,'DiscrimType', 'linear');
%mdl = fitcknn(scoreTrain2,Ytrain);
%mdl = fitcnb(scoreTrain2,Ytrain);
%mdl = fitcensemble(scoreTrain2,Ytrain);
% Neural Network (generic)
mdl = fitcnet(scoreTrain2, Ytrain, ...
           'LayerSizes', [20], ...   % single hidden layer
            'Standardize', true);
%% Test Set PCA
scoreTest2 = (Xtest - mu_pca) * coeff(:,1:2);

%% Predict & Accuracy
Ypred = predict(mdl, scoreTest2);

Accuracy = sum(categorical(Ytest) == categorical(Ypred)) ...
           * 100 / numel(Ytest)
